cmake_minimum_required(VERSION 3.25)
project(EvenSpectrum VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
set(SMTG_RUN_VST_VALIDATOR OFF CACHE BOOL "" FORCE)
if(NOT VST3_SDK_ROOT)
    message(FATAL_ERROR "Set VST3_SDK_ROOT to VST3 SDK 3.8.1 with base, pluginterfaces, public.sdk and cmake submodules.")
endif()
add_subdirectory("${VST3_SDK_ROOT}" sdk)
if(MINGW)
    # This plugin does not use the SDK's optional data-exchange utility,
    # whose aligned allocation helper requires MSVC on Windows.
    get_target_property(sdk_sources sdk SOURCES)
    list(FILTER sdk_sources EXCLUDE REGEX "/utility/dataexchange\\.cpp$")
    set_property(TARGET sdk PROPERTY SOURCES "${sdk_sources}")
    if(TARGET validator)
        target_link_options(validator PRIVATE -municode -static)
    endif()
endif()
set(public_sdk_SOURCE_DIR "${VST3_SDK_ROOT}/public.sdk")
smtg_add_vst3plugin(EvenSpectrum Source/VstPlugin.cpp)
target_include_directories(EvenSpectrum PRIVATE "${VST3_SDK_ROOT}")
target_link_libraries(EvenSpectrum PRIVATE sdk)
if(MINGW)
    target_link_options(EvenSpectrum PRIVATE -static)
    set_target_properties(EvenSpectrum PROPERTIES
        LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/VST3/${CMAKE_BUILD_TYPE}/EvenSpectrum.vst3/Contents/x86_64-win")
endif()
enable_testing()
add_executable(engine_tests Tests/EngineTests.cpp)
if(MINGW)
    target_link_options(engine_tests PRIVATE -static)
endif()
add_test(NAME engine_tests COMMAND engine_tests)
